    <script>
      document.addEventListener("DOMContentLoaded", () => {
        const categoryLinks = document.querySelectorAll(".category-link");
        const filterableCards = document.querySelectorAll(".product-card, .offer-card, .deal-card");
        const sectionTitle = document.querySelector("#produkte .section-header h2");
        const categoryNames = {
          zelte: "Zelte & Schlafsysteme",
          kochen: "Kochen & Feuerstellen",
          licht: "Lampen & Power",
          kleidung: "Outdoor-Kleidung",
          zubehoer: "Zubehör & Werkzeuge"
        };

        function applyCategory(category) {
          let visibleCards = 0;

          filterableCards.forEach((card) => {
            const matches = category === "all" || card.dataset.category === category;
            card.classList.toggle("is-hidden", !matches);
            if (matches) visibleCards += 1;
          });

          categoryLinks.forEach((link) => {
            const isActive = link.dataset.category === category;
            link.classList.toggle("active", isActive);
          });

          if (sectionTitle) {
            sectionTitle.textContent = category === "all"
              ? "Top-Produkte"
              : categoryNames[category] || "Top-Produkte";
          }

          if (visibleCards === 0 && sectionTitle) {
            sectionTitle.textContent = "Keine Produkte in dieser Kategorie";
          }
        }

        categoryLinks.forEach((link) => {
          link.addEventListener("click", (event) => {
            event.preventDefault();
            const category = link.dataset.category || "all";
            applyCategory(category);
            document.getElementById("produkte")?.scrollIntoView({ behavior: "smooth", block: "start" });
          });
        });

        applyCategory("all");
      });
    </script>
